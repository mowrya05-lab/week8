
import java.util.*;
import java.time.*;

abstract class LibraryItem {
    protected String title;

    LibraryItem(String title) {
        this.title = title;
    }

    abstract int getLoanDays();

    String getTitle() {
        return title;
    }

    LocalDate getDueDate(LocalDate currentDate) {
        return currentDate.plusDays(getLoanDays());
    }
}

class Book extends LibraryItem {
    Book(String title) {
        super(title);
    }

    int getLoanDays() {
        return 14;
    }
}

class DVD extends LibraryItem {
    DVD(String title) {
        super(title);
    }

    int getLoanDays() {
        return 7;
    }
}

class Magazine extends LibraryItem {
    Magazine(String title) {
        super(title);
    }

    int getLoanDays() {
        return 3;
    }
}

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = Integer.parseInt(sc.nextLine().trim());

        LibraryItem[] items = new LibraryItem[n];
        LocalDate currentDate = LocalDate.of(2023, 10, 26);

        for (int i = 0; i < n; i++) {
            String line = sc.nextLine().trim();
            int space = line.indexOf(' ');

            String type = line.substring(0, space);
            String title = line.substring(space + 1)
                               .replace("\"", "");

            switch (type.toUpperCase()) {
                case "BOOK":
                    items[i] = new Book(title);
                    break;
                case "DVD":
                    items[i] = new DVD(title);
                    break;
                case "MAGAZINE":
                    items[i] = new Magazine(title);
                    break;
                default:
                    throw new IllegalArgumentException(
                        "Invalid item type");
            }
        }

        for (LibraryItem item : items) {
            System.out.println(item.getTitle() + ": "
                + item.getDueDate(currentDate));
        }

        sc.close();
    }
}
